#
#   cd device/rainx/theloop/sepolicy
#   grep -v 'atci-audio_socket\|loop-control_device\|mtk-adc-cali_device\|rild-dongle_socket\|hal_keymint_strongbox_exec' \
#     ../../../../vendor/rainx/theloop/proprietary/vendor/etc/selinux/vendor_file_contexts \

#   awk '!/^#/ && NF>=2 {print $1}' \
#     ../../../../system/sepolicy/vendor/file_contexts vendor/file_contexts \
#     | sort -u > /tmp/dedup_existing.txt
#   awk 'NR==FNR{dup[$1]=1; next} /^#/||NF<2 {print; next} {if (!($1 in dup)) print}' \
#     /tmp/dedup_existing.txt vendor_mtk_prebuilt/file_contexts.body.raw \
#     > vendor_mtk_prebuilt/file_contexts.body
#   cat vendor_mtk_prebuilt/HEADER.txt vendor_mtk_prebuilt/file_contexts.body \
#     > vendor_mtk_prebuilt/file_contexts
#   rm vendor_mtk_prebuilt/file_contexts.body.raw
